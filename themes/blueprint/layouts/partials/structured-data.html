{{- /* Structured data for search engines: each post as a BlogPosting, the home page as a WebSite. */ -}} {{- $page := .page -}} {{- $data := false -}} {{- if and $page.IsPage (eq $page.Section "post") -}} {{- $data = dict "@context" "https://schema.org" "@type" "BlogPosting" "headline" $page.Title "description" .description "url" $page.Permalink "mainEntityOfPage" $page.Permalink "datePublished" ($page.Date.Format "2006-01-02T15:04:05Z07:00") "dateModified" ($page.Lastmod.Format "2006-01-02T15:04:05Z07:00") "image" (slice (.image | absURL)) "wordCount" $page.WordCount "inLanguage" $page.Site.Language.Lang -}} {{- with $page.Site.Params.author }}{{ $data = merge $data (dict "author" (dict "@type" "Person" "name" .)) }}{{ end -}} {{- with $page.Params.tags }}{{ $data = merge $data (dict "keywords" (delimit . ", ")) }}{{ end -}} {{- else if $page.IsHome -}} {{- $data = dict "@context" "https://schema.org" "@type" "WebSite" "name" $page.Site.Title "description" .description "url" $page.Permalink "inLanguage" $page.Site.Language.Lang -}} {{- end -}} {{ with $data }}{{ end }}
